Public Property Get ksep_total_fr() As Double
' общая сепарация УЭЦН параметр только для чтения
' для задания вручную надо задать естественную сепарацию и сепарацию газосепаратора
    ksep_total_fr = ksep_nat_fr + (1 - ksep_nat_fr) * ksep_gassep_fr
End Property

                          d_cas_mm:=0, _
                          angle_deg:=90)
'    isGasSep = True
    separation_mode = byCorrealation
    ksep_nat_fr = 0.5
    ksep_gassep_fr = 0.5
    ksep_gassep_max_d_ = 0.9
    ksep_q_max_m3day_ = 500
    pkv_period_ratio = 1
    h_intake_m = 0.1
    h_perf_m = 1
    p_ksep_atma = 50
    t_ksep_C = 60
    natsep.calibr_li = 2
    natsep.calibr_st = 1
    d_tub_m = 0.073
    ksep_liq_d = 0
    ksep_liquid_man_d = 0
End Sub
Public Function calc_separation(p_atma As Double, t_C As Double, _
                                Optional p_cas_atma As Double = 10)
' расчет сепарации и соответствующих параметров флюидов
' для момента работы скважины
' должен вызываться незавимо от расчета УЭЦН, так как при расчете сверху вниз
'   необходимо заранее рассчитать параметры флюида выше насоса, чтобы правильно посчитать НКТ
' p_atma  - давление при которой идет сепарация
' T_C - температура при которой идет сепарация
    
    
    Select Case separation_mode
        Case "by_correlation"
            ' все по сепарации по корреляциям
            p_ksep_atma = p_atma
            t_ksep_C = t_C
            'Call fluid_.calc_PVT(p_ksep_atma, t_ksep_C)
            ksep_nat_fr = estimate_nat_separation_d(p_ksep_atma, t_ksep_C) ' оценим естественную сепарацию
            ksep_liq_d = 0
        Case "by_correlation_pressure_manual"
            ' берем значение ручное, давления как есть
            p_ksep_atma = psep_man_atma
            t_ksep_C = tsep_man_C
            'Call fluid_.calc_PVT(p_ksep_atma, t_ksep_C)
                ' оценим естественную сепарацию
            ksep_nat_fr = estimate_nat_separation_d(p_ksep_atma, t_ksep_C)
            ksep_liq_d = 0
        Case "value_manual"
            ' берем и коэффициенты и давления как было задано при установке значений
            p_ksep_atma = p_atma
            t_ksep_C = t_C
            ksep_nat_fr = ksep_nat_man_d
            ksep_liq_d = ksep_liquid_man_d
        Case "fully_manual"
            ' берем и коэффициенты и давления как было задано при установке значений
            p_ksep_atma = psep_man_atma
            t_ksep_C = tsep_man_C
            ksep_nat_fr = ksep_nat_man_d
            ksep_liq_d = ksep_liquid_man_d
    End Select
    
    
    Call estimate_feeds_after_separation_calc(p_ksep_atma, t_ksep_C)
    
    Select Case separation_mode
        Case "by_correlation"
            ksep_gassep_fr = estimate_GS_separation_d(p_ksep_atma, t_ksep_C)
        Case "by_correlation_pressure_manual"
                ' берем значение ручное, давления как есть
                ' оценим коэффициент сепарации газосепаратора
                ksep_gassep_fr = estimate_GS_separation_d(p_ksep_atma, t_ksep_C)
        Case "value_manual"
                ' берем и коэффициенты и давления как было задано при установке значений
            ksep_gassep_fr = ksep_gassep_man_d
        Case "fully_manual"
                ' берем и коэффициенты и давления как было задано при установке значений
            ksep_gassep_fr = ksep_gassep_man_d
    End Select
    
    ksep_gassep_fr = ksep_gassep_fr * ksep_ann
    ' на этом этапе оценили коэффициенты сепарации
    ' далее надо посчитать параметры потоков
    
    If ksep_liq_d <= 0 Or ksep_liq_d > 1 Then
        ' нормальный режим работы скважины
        Set fluid_tub = fluid_gs.clone()
        Call fluid_tub.mod_after_separation(p_ksep_atma, t_ksep_C, ksep_gassep_fr, True)
        Call fluid_tub.calc_PVT(p_atma, t_C)
        q_gas_annulus_after_full_sep_sm3day = fluid.q_gas_insitu_sm3day - fluid_tub.q_gas_insitu_sm3day
    
        Set fluid_cas = fluid.clone() ' тут надо будет доделать чтобы считать затруб с газом
        Call fluid_cas.mod_after_separation(p_cas_atma, t_ksep_C, 0.99, gas_goes_into_solution:=False)
        fluid_cas.fw_perc = 0
        fluid_cas.q_gas_free_sm3day = q_gas_annulus_after_full_sep_sm3day
        fluid_cas.q_liq_sm3day = 0
        
        
        If pkv_period_ratio > 0 And pkv_period_ratio < 1 Then
            Dim ql_from_cas As Double
            Dim qg_to_cas As Double
            
            Call fluid_from_annular.calc_PVT(p_atma, t_C)
            Call fluid_cas.calc_PVT(p_atma, t_C)
            
            If q_gas_annulus_after_full_sep_sm3day > 0 Then
                ' тут жидкость вниз газ вверх в затрубе
                ql_from_cas = fluid_from_annular.qliq_rc_m3day  ' дебит жидкости который мы забираем из затруба в момент работы
                qg_to_cas = fluid_cas.q_gas_rc_m3day
                q_cas_rc_m3day = qg_to_cas - ql_from_cas
            Else
                q_cas_rc_m3day = -fluid_from_annular.q_mix_rc_m3day
    
            End If
        Else
            Call fluid_cas.calc_PVT(p_atma, t_C)
            q_cas_rc_m3day = q_gas_annulus_after_full_sep_sm3day * fluid_cas.bg_m3m3
        End If
    Else
        ' фонтан через затруб с разделение потоков
        Dim dfl As Dictionary
        Set dfl = fluid.mod_split(ksep_total_fr, ksep_liq_d, ksep_liq_d, p_ksep_atma, t_ksep_C, True)
        Set fluid_tub = dfl("feed1")
        Set fluid_cas = dfl("feed2")
    End If
    calc_separation = ksep_total_fr
End Function
Public Sub set_gassep_param(Optional OD_mm As Double = 125, _
                       Optional Qnom_m3day As Double = 250, _
                       Optional h_mes_top_m As Double = 0, _
                       Optional angle_deg As Double = 90, _
                       Optional d_cas_mm As Double = 0) ', _
'                       Optional length_m As Double = 1)
' метод инициализации параметров газосепаратора
' в зависимости от габарита
' OD_mm     - определяет габарит
' h_mes_top_m - глубина подвеса в скважине, может пригодится когда то, пока про запас
' Angle_deg - угол установки к горизонтали 90 - вертикально
' d_cas_mm   - диаметр эксплуатационной колоны в месте установки
    d_od_mm_ = OD_mm
'    Me.angle_deg = angle_deg
    If d_cas_mm = 0 Then
        Me.d_cas_m = (OD_mm + 10) / 1000
    Else
        Me.d_cas_m = d_cas_mm / 1000
    End If
'    h_mes_top_ = h_mes_top_m
'    length_m_ = length_m
    ksep_q_max_m3day_ = Qnom_m3day * 2
End Sub
Public Sub init_json(json As String)
    Dim d As Dictionary
    Set d = ParseJson(json)
    Call init_dictionary(d)
End Sub
 
Public Sub init_dictionary(dict As Dictionary)
Dim ESP_ID As Long
Dim head_nom_m As Double
Dim msg As String
On Error GoTo err1:
   With dict
        If .Exists("calc_mode") Then
            separation_mode = .Item("calc_mode")
        Else
            separation_mode = 0
        End If
        
        If .Exists("gassep_type") Then
            gassep_type = .Item("gassep_type")
            If .Exists("gassep_qnom_sm3day") Then ksep_q_max_m3day_ = .Item("gassep_qnom_sm3day") * 2
        Else
            gassep_type = 0
        End If
        
        If .Exists("natsep_model") Then
            natsep_type = .Item("natsep_model")
        Else
            natsep_type = 0
        End If
        
        If .Exists("p_sep_manual_atma") Then psep_man_atma = .Item("p_sep_manual_atma")
        If .Exists("t_sep_manual_C") Then tsep_man_C = .Item("t_sep_manual_C")
        If .Exists("ksep_gassep_man_d") Then ksep_gassep_man_d = .Item("ksep_gassep_man_d")
        If .Exists("ksep_nat_man_d") Then ksep_nat_man_d = .Item("ksep_nat_man_d")
        If .Exists("ksep_liquid_man_d") Then ksep_liquid_man_d = .Item("ksep_liquid_man_d")
        If .Exists("d_intake_mm") Then d_od_mm_ = .Item("d_intake_mm")
        If .Exists("d_cas_mm") Then d_cas_m = .Item("d_cas_mm") / 1000
        If .Exists("d_tub_mm") Then d_tub_m = .Item("d_tub_mm") / 1000
        If .Exists("h_intake_m") Then h_intake_m = .Item("h_intake_m")
        If .Exists("h_perf_m") Then h_perf_m = .Item("h_perf_m")
        If .Exists("pkv_period_ratio") Then pkv_period_ratio = .Item("pkv_period_ratio")
        If natsep_type = 3 And pkv_period_ratio = 1 Then
            natsep_type = 0
        End If
        
        Dim prmd As Dictionary
        If .Exists("param") Then
            Set prmd = .Item("param")
            If prmd.Exists("calibr_li") Then natsep.calibr_li = prmd("calibr_li")
            If prmd.Exists("calibr_st") Then natsep.calibr_st = prmd("calibr_st")
            If prmd.Exists("gas_goes_into_solution") Then gas_goes_into_solution = prmd("gas_goes_into_solution")
        End If
   End With
    
   Set natsep = New CWellNaturalSeparation
   natsep.d_intake_m = d_od_m
   natsep.d_cas_m = d_cas_m
   natsep.d_tub_m = d_tub_m
   natsep.pkv_ratio = pkv_period_ratio
   natsep.h_intake_m = h_intake_m
   natsep.h_perf_m = h_perf_m
  
   Exit Sub
err1:
   Dim errmsg As String
   errmsg = "Error:CESPGasSep.init_dictionary: init error " & sDELIM & Err.Description
   add_log_msg errmsg
   Err.Raise kErrESPcalc, Err.source, errmsg
   'Resume Next
End Sub

Private Function make_feed_from_annulus(p_atma As Double, t_C As Double, ksep_ann As Double) As CPVT
    Dim fl As CPVT
    Set fl = fluid.clone
    ' для флюида из затруба уберем весь газ из потока (ксеп = 1)
    Call fl.mod_after_separation(p_atma, t_C, ksep_ann, gas_goes_into_solution)
    ' учтем, что из затруба объем жидкости определяется пкв отношением
    fl.q_liq_sm3day = fluid.q_liq_sm3day * 1 / (pkv_period_ratio) * (1 - pkv_period_ratio)
    ' может быть можно было бы учесть, что поменяется обводненность - но пока не будем
    fl.rp_m3m3 = fl.rsb_m3m3
    Set make_feed_from_annulus = fl
End Function
Private Sub estimate_feeds_after_separation_calc(p_atma As Double, t_C As Double)
    ' оценка параметров потока в том числе для ПКВ режима
    Dim qliq_from_res As Double
    Dim qliq_from_annulus As Double
    Dim vg_downflow_eps As Double
    Dim vg_downflow_limit As Double
    Dim ksep_ann2 As Double
    
    Set fluid_gs = fluid.clone
    If pkv_period_ratio > 0 And pkv_period_ratio < 1 Then
        ' если тут значит работаем в ПКВ режиме и надо посчитать ксеп дополнительно
        ' в ПКВ возможно резкое изменение коэффициента сепарации при переходе границы по дебиту
        ' при которой газ в противотоке может подниматься или не может
        ' резкая граница неудобна для расчетов и вряд ли реализуется на практике
        ' поэтому сгладим ее в расчете задав позицию границы (limit) и ее размытие (eps)
        '    эти параметры можно настроить на реальные данные если такие будут
        vg_downflow_eps = 0.1
        vg_downflow_limit = -0.05
        ' оценим дебиты жидкости из пласта и затруба при работе ПКВ
        qliq_from_res = fluid.q_liq_sm3day
        qliq_from_annulus = qliq_from_res * 1 / (pkv_period_ratio) * (1 - pkv_period_ratio)
        
        ' для начала предположим, что из затруба только жидкость идет вниз (без учета изменения обводненности)
        ksep_ann = 0.99
        Set fluid_from_annular = make_feed_from_annulus(p_atma, t_C, ksep_ann)
        ' уточним сепарацию жидкости из затруба
        Call natsep.init_pvt(fluid_from_annular, p_atma, t_C)
        If natsep.vg_downflow_msec > vg_downflow_limit + vg_downflow_eps Then
             ' скорость газа положительная значит газ может убежать от нисходящей жидкости
             ' и тогда из затруба будет идти чистая жидкость
            ksep_ann = 0.99
        Else
             ' скорость газа ноль или отрицательная - значит весь газ пойдет в насос (почти)
             If natsep.vg_downflow_tub_msec > vg_downflow_limit Then
                 ' если выше насоса в зазоре между НКТ и э/к скорость газа отрицательная - сепарации точно не будет
                 ksep_ann2 = 0.1
             Else
                 ' тут можно оценить количество газа в зазоре между ЭЦН и э\к и транслировать в ксеп
                 ' но для этого может потребоваться длительность периода абсолютная
                 ksep_ann2 = 0.01
             End If
             ' далее сгладим ксеп чтобы не было резкого скачка
             If natsep.vg_downflow_msec > vg_downflow_limit Then
                 ksep_ann = (0.99 - ksep_ann2) * (1 - (vg_downflow_limit + vg_downflow_eps - natsep.vg_downflow_msec) / (vg_downflow_eps - vg_downflow_limit)) + ksep_ann2
             Else
                 ksep_ann = ksep_ann2
             End If
        End If
        ' пересчитаем потоке с новой оценкой ксеп
        Set fluid_from_annular = make_feed_from_annulus(p_atma, t_C, ksep_ann)
        ' поправим ранее рассчитанный коэффициент натуральной сепарации
        ' используем оценку сепарации нисходящего потока как множитель для естественной сепарации
        ' помним, что если в противотоке газ всплыть не может, то натуральной сепарации не будет
        ksep_nat_fr = ksep_nat_fr * ksep_ann
        Call fluid_gs.mod_after_separation(p_atma, t_C, ksep_nat_fr, gas_goes_into_solution)
        ' смешаем флюиды из затруба и из пласта в один который пойдет в приемную сетку насоса
        Set fluid_gs = feed_mod_mix_obj(fluid_gs, fluid_from_annular)
    Else
        ' если ПКВ нет то просто учитываем сепарацию
        Call fluid_gs.mod_after_separation(p_atma, t_C, ksep_nat_fr, gas_goes_into_solution)
        Call fluid.calc_PVT(p_atma, t_C)
        Call fluid_gs.calc_PVT(p_atma, t_C)
        ksep_ann = 1
        Set fluid_from_annular = fluid.clone
        fluid_from_annular.q_liq_sm3day = 0
        fluid_from_annular.q_gas_free_sm3day = 0
    End If
    
    q_gas_annulus_after_nat_sep_sm3day = fluid.q_gas_insitu_sm3day - fluid_gs.q_gas_insitu_sm3day
End Sub
Private Function estimate_GS_separation_d(p_atma As Double, t_C As Double, Optional f_Hz As Double = 50)
' расчет коэффициента сепарации газосепаратора
' предполагается линейная зависимость от дебита
' q_mix_rc_m3day - расход ГЖС поступающей на вход с учетом естественной сепарации с рабочих условиях
' потом хорошо бы уточнить и приспособить для разных видов конструкции
    Dim KS As Double
    Dim q_mix_ As Double
    Dim qmax As Double
    
    Call fluid_gs.calc_PVT(p_atma, t_C)
    Select Case gassep_type
        Case 0
            ' отключаем сепарацию газосепаратором
            estimate_GS_separation_d = 0
        Case -1
            q_mix_ = fluid_gs.q_mix_rc_m3day
            qmax = ksep_q_max_m3day_
            
            If q_mix_ < qmax Then
                KS = (1 - q_mix_ / qmax) * ksep_gassep_max_d_
            Else
                KS = 0
            End If
            estimate_GS_separation_d = KS
        Case Else
            estimate_GS_separation_d = gassep.my_sepfactor(gassep_type, fluid_gs.gas_fraction_d * 100, fluid_gs.q_liq_sm3day, f_Hz * 60) / 100
   End Select
    
End Function
Private Function estimate_nat_separation_d(p_atma As Double, t_C As Double) As Double
    ' оцениваем естественную сепарацию и свойства потока в насос после сепарации и расход газа в затруб после сепарации
    Dim qliq As Double
    Dim qgas As Double
    
    Call natsep.init_pvt(fluid, p_atma, t_C)
    Select Case natsep_type
        Case -1
            ' натуральная сепарация отключена
            estimate_nat_separation_d = 0
        Case 0, 1, 2
            qliq = fluid.qliq_rc_m3day
            
            If pkv_period_ratio > 0 And pkv_period_ratio < 1 Then
                ' ПКВ режим можно активировать как для упрощенной так и для механистической корреляции Маркеса
                qliq = qliq / (pkv_period_ratio)
            End If
            Select Case natsep_type
                Case 0
                    ' упрощенный расчет по Маркесу
                    Call natsep.ksep_simplified_Marquez(qliq, fluid.q_gas_rc_m3day)
                Case 1
                    ' механистический расчет по Маркесу
                    Call natsep.ksep_mechanistic_Marquez(qliq, fluid.q_gas_rc_m3day)
                Case 2
                    ' механистический расчет по Маркесу для эцн ниже перфорации
                    Call natsep.ksep_mechanistic_Marquez_reverse(qliq, fluid.q_gas_rc_m3day)
                 End Select
            estimate_nat_separation_d = natsep.ksep_work
            ' оценим свойства потока идущего в насос
            ' оценим расход газа в затруб после естественной сепарации
        Case 3
            '  режим расчета ПКВ
            '  режим 3 устарел - в перспективе надо будет убрать возможность активации
             Call natsep.ksep_mechanistic_pkv(fluid.qliq_rc_m3day, fluid.q_gas_rc_m3day)
             estimate_nat_separation_d = natsep.ksep_work
     End Select
End Function
