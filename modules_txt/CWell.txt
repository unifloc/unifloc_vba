Public calibr_grav As Double
Public calibr_fric As Double

Public separ_calc As Boolean            ' флаг что расчет сепарации был проведен
Public ksep_total_d As Double           ' расчетное значение общего коэффициента сепарации для вывода
Public gas_fraction_pump_d As Double    ' расчетная доля газа в насосе после сепарации для вывода

Public t_buf_C As Double
Public p_cas_atma As Double

Private Sub Class_Initialize()
    fast = False
    d_glv_mm = 0   ' 0 означает расчет выключен
    calibr_glv = 1
    d_gas_surf_mm = 0
    calibr_surf = 1
    calibr_grav = 1
    calibr_fric = 1
    separ_calc = False
End Sub

Public Sub init_esp_well(Optional ByVal trajectory As String = "1500", _
                         Optional ByVal d_tub_mm As String = "62", _
                         Optional ByVal d_cas_mm As String = "125", _
                         Optional ByVal t_model As String = "80", _
                         Optional ByVal h_perf_m As Double = 1500, _
                         Optional ByVal h_esp_m As Double = 1200, _
                         Optional ByVal esp_json As String = "", _
                         Optional ByVal ksep_json As String = "", _
                         Optional ByVal d_int_mm As Double = 110, _
                         Optional ByVal IPR_json As String = "", _
                         Optional ByVal calibr_grav As Double = 1, _
                         Optional ByVal calibr_fric As Double = 1, _
                         Optional ByVal flow_corr As Integer = 0)
    Dim ksep As Double
    Dim d_cas_mm_ As Double
    Dim d_tub_mm_ As Double
    
    Dim dsep As Dictionary
    Call create_ipr(IPR_json)
    
    ' оценим длину насоса в самом начале есть насос для расчета подключен
    ' необходимо, чтобы красиво нарисовать итоговые графики
    esp_length = 0
    If Len(esp_json) > 1 Then
        Call esp.init_json(esp_json)
        If esp.length_m > 0 Then esp_length = esp.length_m
        If h_perf_m < h_esp_m + esp_length Then
            ' заглушка - пока расчет ЭЦН ниже перфорации не поддерживается, хотя
            ' в перспективе можно реализовать
            Err.Raise kErrESPcalc, , "error " & "pump intake below perf not supported. " & str(h_perf_m) & " < " & str(h_esp_m + esp_length)
            Exit Sub
        End If
    End If
    ' создадим основные трубы
    ' создаются отдельной процедурой общей для скважины с ЭЦН и с газлифтом
    Call create_main_pipes(trajectory, _
                           d_tub_mm, d_cas_mm, _
                           t_model, _
                           h_perf_m, h_esp_m, _
                           calibr_grav, _
                           calibr_fric, _
                           flow_corr, _
                           esp_length, wall_thick_mm:=5, esp_calc:=True)
    ' создадим объекты для сепарации газа на приеме ЭЦН
    If ksep_json = "" Then
        separ_calc = False
    Else
        separ_calc = True
        ' есть сепарация газа - пытаемся ее учесть
        If Mid(ksep_json, 1, 1) <> "{" Then
            ' если параметры сепарации в виде числа
            ' то создаем объект сепарации с постоянным значением к-та сепарации
            ksep = Cdbl_(ksep_json)
            ' берем диаметры на нужной глубине из заранее созданных труб
            d_cas_mm_ = pipe_below_al.d_hmes_m(h_esp_m + esp_length) * 1000
            d_tub_mm_ = pipe_tubing_above_al.d_hmes_m(h_esp_m + esp_length) * 1000
            ' создаем объект сепарации
            ' газосепаратора нет, расчет 2 с постоянным значение ксеп
            ksep_json = encode_ESP_separation( _
                                    calc_mode:=2, _
                                    p_sep_manual_atma:=0, _
                                    t_sep_manual_C:=0, _
                                    natsep_model:=0, _
                                    d_intake_od_mm:=d_int_mm, _
                                    d_cas_id_mm:=d_cas_mm_, _
                                    d_tub_od_mm:=d_tub_mm_, _
                                    pkv_period_ratio:=0, _
                                    ksep_gassep_man_d:=0, _
                                    ksep_nat_man_d:=ksep, _
                                    ksep_liquid_man_d:=0, _
                                    h_intake_m:=0, _
                                    h_perf_m:=0, _
                                    gassep_type:=0, _
                                    gassep_qnom_sm3day:=100, _
                                    param:="")
        End If
        ' по любому здесь уже доступна строка объекта сепарации - парсим ее
        Set dsep = ParseJson(ksep_json)
        Call separ.init_dictionary(dsep)
    End If
End Sub
' инициализация конструкции газлифтной скважины
Public Sub init_gl_well(Optional ByVal trajectory As String = "1500", _
                        Optional ByVal d_tub_mm As String = "62", _
                        Optional ByVal d_cas_mm As String = "125", _
                        Optional ByVal t_model As String = "80", _
                        Optional ByVal h_perf_m As Double = 1500, _
                        Optional ByVal h_glv_m As Double = 1200, _
                        Optional ByVal glv_json As String = "", _
                        Optional ByVal IPR_json As String = "", _
                        Optional ByVal calibr_grav As Double = 1, _
                        Optional ByVal calibr_fric As Double = 1, _
                        Optional ByVal flow_corr As Integer = 0)
    Call create_ipr(IPR_json)
    Call create_main_pipes(trajectory, _
                           d_tub_mm, d_cas_mm, _
                           t_model, _
                           h_perf_m, h_glv_m, _
                           calibr_grav, _
                           calibr_fric, _
                           flow_corr)
End Sub
Private Sub create_ipr(ByVal IPR_json As String)
    If Len(IPR_json) > 0 Then
        Call ipr_vogel.init_json(IPR_json)
        Set ipr_vogel.fluid = fluid
    End If
End Sub
Private Sub create_main_pipes(trajectory As String, _
                              d_tub_mm As String, _
                              d_cas_mm As String, _
                              t_model As String, _
                              h_perf_m As Double, _
                              h_al_m As Double, _
                              calibr_grav As Double, _
                              calibr_fric As Double, _
                              flow_corr As Integer, _
                     Optional al_length As Double = 0, _
                     Optional wall_thick_mm As Double = 5, _
                     Optional esp_calc As Boolean = False)
    ' создаем трубу ниже мех. добычи для расчета снизу вверх
    Set pipe_below_al = new_pipeline4(trajectory, _
                                      d_cas_mm, _
                                      t_model, _
                                      flow_corr, _
                                      calibr_grav, _
                                      calibr_fric, _
                                      h_al_m + al_length, h_perf_m, h_al_m, _
                                      calc_along_coord:=False, _
                                      flow_along_coord:=False)
    Set pipe_below_al.fluid = fluid
    ' создаем трубу НКТ выше мех. добычи для расчета снизу вверх
    Set pipe_tubing_above_al = new_pipeline4(trajectory, _
                                             d_tub_mm, _
                                             t_model, _
                                             flow_corr, _
                                             calibr_grav, _
                                             calibr_fric, _
                                             0, _
                                             h_al_m, h_al_m, h_al_m, _
                                             calc_along_coord:=False, _
                                             flow_along_coord:=False)
    Set pipe_tubing_above_al.fluid = fluid
    'построим распределение давления для газа
    'планируем расчет газа в затрубе сверху вниз
    Set pipe_casing_gas = new_pipeline4(trajectory, _
                                        pipe_diam_subtract(d_cas_mm, d_tub_mm, wall_thick_mm), _
                                        t_model, _
                                        10, _
                                        calibr_grav, _
                                        calibr_fric, _
                                        0, h_al_m + al_length, h_al_m, h_al_m, _
                                        calc_along_coord:=True, _
                                        flow_along_coord:=False)
    Set pipe_casing_gas.fluid = fluid
    
    If esp_calc Then
    
        ' если только жидкость в затрубе
        Set pipe_casing_liq = new_pipeline4(trajectory, _
                                            pipe_diam_subtract(d_cas_mm, d_tub_mm, wall_thick_mm), _
                                            t_model, _
                                            flow_corr, _
                                            calibr_grav, _
                                            calibr_fric, _
                                            0, h_al_m + al_length, h_al_m, 0, _
                                            calc_along_coord:=False, _
                                            flow_along_coord:=False)
    End If
    
    Me.calibr_grav = calibr_grav
    Me.calibr_fric = calibr_fric
    
    t_surf_C = pipe_casing_gas.ambient_formation.amb_temp_C(0)
End Sub
Public Function calc_esp_well_from_pwf(ByVal p_wf_atma As Double, _
                                       ByVal t_wf_C As Double, _
                              Optional ByVal p_cas_atma As Double = 10, _
                              Optional ByVal esp_freq_Hz As Double = 50, _
                              Optional ByVal pkv_ratio As Double = 0, _
                              Optional ByVal t_crit_C As Double = 0)
    ' Расчет распределения давления и температуры в скважине с ЭЦН
    '
    Dim res1 As PTtype
    Dim i As Integer
    Dim prm As PARAMCALC
    Dim pipe_curves_fast As CALC_RESULTS
    If fast Then
        pipe_curves_fast = noCurves
    Else
        pipe_curves_fast = mainCurves
    End If
    
    calc_esp_well_from_pwf = False ' перед стартом расчета предполагаем худшее на всякий случай
        
    Me.p_wf_atma = p_wf_atma
    Me.t_wf_C = t_wf_C
    ' 1. Оценим дебит жидкости
    '    можем вначале это сделать так как это расчет снизу вверх
    If ipr_vogel.p_res_atma > 0 Then
        ' оценим дебит жидкости по IPR если задано
        q_liq_sm3day = ipr_vogel.calc_q_liq_sm3day(p_wf_atma)
    Else
        ' довольствуемся дебитом заданном во входном потоке
        q_liq_sm3day = fluid.q_liq_sm3day
    End If
    fluid.q_liq_sm3day = q_liq_sm3day
    '2. рассчитаем участок ниже мех добычи (газлифтного клапана или ЭЦН)
    Set pipe_below_al.fluid = fluid
    res1 = pipe_below_al.calc_dPipe(p_wf_atma, t_wf_C, pipe_curves_fast)
    p_intake_atma = res1.p_atma
    t_intake_C = res1.t_C
    q_gas_cas_sm3day = 0
    If pipe_curves_fast = mainCurves Then Set crv_p_below_al = pipe_below_al.curve("c_P")
    
    '3. если надо (если установлен ЭЦН) оценим сепарацию газа на приеме насоса
    If esp.stage_num > 0 Then
        separ_calc = True
        Set separ.fluid = fluid
        If pkv_ratio > 0 And pkv_ratio < 1 Then
            separ.pkv_period_ratio = pkv_ratio
        End If
        Call separ.calc_separation(p_intake_atma, t_intake_C)
        Set fluid_al_ = separ.fluid_tub
        Set fluid_cas_ = separ.fluid_cas
        ksep_total_d = separ.ksep_total_fr
        q_gas_cas_sm3day = separ.q_gas_annulus_after_full_sep_sm3day
        ' если ЭЦН есть рассчитаем перепад давления
        Set esp.fluid = fluid_al_
        Call fluid_al_.calc_PVT(p_intake_atma, t_intake_C)
        gas_fraction_pump_d = fluid_al_.gas_fraction_d
        Call esp.calc_ESP(p_intake_atma, t_intake_C, f_Hz:=esp_freq_Hz)
        ' вытащим значения давления на выкиде
        p_dis_atma = esp.p_dis_atma
        t_dis_C = esp.t_dis_C
    Else
        ' если данных по ЭЦН нет - тогда это фонтанирующая скважина
        separ_calc = False
        Set fluid_al_ = fluid.clone
        Set fluid_cas_ = fluid.clone
        p_dis_atma = p_intake_atma
        t_dis_C = t_intake_C
        q_gas_cas_sm3day = 0
        gas_fraction_pump_d = 0
        fluid_cas_.q_liq_sm3day = 0
        fluid_cas_.q_gas_free_sm3day = 0
    End If
    
    ' 4. построим распределение давления в затрубе
    If p_cas_atma < 1 Then p_cas_atma = 1
    prm = pipe_casing_gas.param
    prm.calc_along_coord = False
    pipe_casing_gas.param = prm
    Set pipe_casing_gas.fluid = fluid_cas_
    Set pipe_casing_liq.fluid = fluid_cas_
    If esp.stage_num > 0 Then
        res1 = pipe_casing_gas.calc_dPipe(p_intake_atma, t_intake_C, mainCurves)
        ' сохраним предельное затрубное давление
        ' такое затрубное которое будет при предположении что весь затруб
        ' заполнен только газом. Его также выведем в результатах
        ' чтобы обеспечить корректный поиск забойного давления по устьевому
        p_cas_limit_atma = res1.p_atma
        If separ.ksep_liq_d <= 0 Or separ.ksep_liq_d > 1 Then
            Call pipe_casing_liq.set_ZNLF  ' учитываем барботаж для жидкости
            res1 = pipe_casing_liq.calc_dPipe(p_intake_atma, t_intake_C, mainCurves)
            Set crv_cas_p = pipe_casing_liq.curve("c_P")  ' готовим кривую давлений для вывода результатов
            ' далее разбираем возможные конфигурации распределения давления в затрубе
            If res1.p_atma < p_cas_atma Then
                ' если давление затруба с жидкостью меньше заданного
                ' то это нормальный вариант когда может существовать динамический уровень или
                ' затруб полностью заполнен газом
                '  продолжаем разбор вариантов
                If p_cas_limit_atma < p_cas_atma Then
                    ' если предельное затрубное давление меньше фактического значит
                    ' попали в вариант когда затрубное давление является ограничением для снижения забойного давления
                    ' как результат покажем заниженное затрубное давление
                    Set crv_cas_p = pipe_casing_gas.curve("c_P")
                    p_cas_atma = p_cas_limit_atma
                Else
                    ' здесь нормальный вариант когда должен существовать динамический уровень
                    ' ищем его как пересечение кривых распределения давления в жидкости и газе
                    ' построенных навстречу друг другу
                    ' построим распределение давления для газа
                    prm.calc_along_coord = True
                    pipe_casing_gas.param = prm
                    res1 = pipe_casing_gas.calc_dPipe(p_cas_atma, t_intake_C, mainCurves)
                    ' найдем динамический уровень и построим совместное распределения для газа и жидкости
                    h_dyn_m = intersection(pipe_casing_liq.curve("c_P"), pipe_casing_gas.curve("c_P"))(1)
                    If h_dyn_m < 0 Then h_dyn_m = 0
                    pipe_casing_liq.curve("c_P").xName = "h_mes,m"
                    pipe_casing_liq.curve("c_P").yName = "p,atma"
                    ' сохраним для отчета кривую распределения давления в затрубе с учетом двух участков
                    For i = 1 To crv_cas_p.num_points
                        Call crv_cas_p.AddPoint(h_dyn_m, crv_cas_p.getPoint(h_dyn_m), True)
                        If crv_cas_p.pointX(i) < h_dyn_m Then
                            Call crv_cas_p.AddPoint(crv_cas_p.pointX(i), pipe_casing_gas.curve("c_P").getPoint(crv_cas_p.pointX(i)))
                        End If
                    Next i
                    Me.p_cas_atma = p_cas_atma
                End If
            Else
                Me.p_cas_atma = res1.p_atma
            End If
        Else
                ' тут может быть фонтан через затруб
            res1 = pipe_casing_liq.calc_dPipe(p_intake_atma, t_intake_C, mainCurves)
            Set crv_cas_p = pipe_casing_liq.curve("c_P")  ' готовим кривую давлений для вывода результатов
            Me.p_cas_atma = res1.p_atma
        End If
    Else
        ' скважина фонтанирует и тогда предполагаем, что в затрубе будет только газ
        ' и расчет газа ведем снизу вверх
        
        res1 = pipe_casing_gas.calc_dPipe(p_intake_atma, t_intake_C, mainCurves)
        Set crv_cas_p = pipe_casing_gas.curve("c_P")  ' готовим кривую давлений для вывода результатов
        Me.p_cas_atma = res1.p_atma
        
    End If
    
    ' 5. расчет распределения давления в НКТ
    '    считаем после затруба, чтобу уже знать динамический уровень
    '    и учесть влияние динамического уровня на распределение температуры
    '    для температурной модели с учетом эмиссии тепла в окружающее пространство
    Set pipe_tubing_above_al.fluid = fluid_al_
    pipe_tubing_above_al.ambient_formation.h_dyn_m = pipe_tubing_above_al.h_vert_h_mes_m(h_dyn_m)
    res1 = pipe_tubing_above_al.calc_dPipe(p_dis_atma, t_dis_C, mainCurves)
    ' сохраним для отчета найденное буферное давление
    ' найдем глубину на которой достигается критическая температура
    Call build_crv(True)
    Dim n_sol As Integer
    Me.t_crit_C = t_crit_C
    h_crit_m = 0
    n_sol = crv_t.FindSolutions(t_crit_C, TS_NO_EXTRPOLATION)
    If n_sol > 0 Then h_crit_m = crv_t.SolutionPointX(1)
    If t_crit_C > crv_t.maxy Then h_crit_m = crv_t.maxx
    p_buf_atma = res1.p_atma
    t_buf_C = res1.t_C
End Function
' расчет газлифтной скважины снизу вверх по забойному давлению
Public Function calc_gl_well_from_pwf(ByVal p_wf_atma As Double, _
                                      ByVal t_wf_C As Double, _
                             Optional ByVal p_cas_atma As Double = 150, _
                             Optional ByVal q_gas_glv_sm3day As Double = 0)
    Dim res1 As PTtype
    
    calc_gl_well_from_pwf = False
    
    Me.p_wf_atma = p_wf_atma
    Me.t_wf_C = t_wf_C
    
    ' оценим дебит жидкости
    If ipr_vogel.p_res_atma > 0 Then
        q_liq_sm3day = ipr_vogel.calc_q_liq_sm3day(p_wf_atma)
    Else
        q_liq_sm3day = fluid.q_liq_sm3day
    End If
    
    ' рассчитаем участок ниже мех добычи (газлифтного клапана)
    Set pipe_below_al.fluid = fluid
    If fast Then
        res1 = pipe_below_al.calc_dPipe(p_wf_atma, t_wf_C, noCurves)
    Else
        res1 = pipe_below_al.calc_dPipe(p_wf_atma, t_wf_C, mainCurves)
    End If
    p_intake_atma = res1.p_atma
    t_intake_C = res1.t_C
    
    ' рассчитаем распределение давления газа в затрубе
    
    If d_glv_mm > 0 Then
        ' если есть диаметр клапана пытаемся его учесть в расчете
        If d_gas_surf_mm > 0 Then
            q_gas_glv_sm3day = GLV_q_gas_two_valves_sm3day(d_glv_mm, d_gas_surf_mm, _
                                                           p_cas_atma, p_intake_atma, _
                                                           fluid.gamma_g, _
                                                           t_surf_C, t_intake_C, _
                                                           pipe_casing_gas, _
                                                           calibr_surf, calibr_glv)
        Else
            res1 = pipe_casing_gas.calc_dPipe(p_cas_atma, t_intake_C, mainCurves)
            q_gas_glv_sm3day = GLV_q_gas_sm3day(d_glv_mm, res1.p_atma, p_intake_atma, fluid.gamma_g, t_intake_C)(0)(0)
        End If
    Else
        res1 = pipe_casing_gas.calc_dPipe(p_cas_atma, t_intake_C, mainCurves)
    End If
    p_gas_intake_atma = res1.p_atma
    t_gas_intake_C = res1.t_C
    
    ' в итого подготовим флюид для расчета выше клапана
    Set fluid_al_ = fluid.clone
    fluid_al_.q_gas_free_sm3day = fluid_al_.q_gas_free_sm3day + q_gas_glv_sm3day
    Me.q_gas_glv_sm3day = q_gas_glv_sm3day
    ' оценим температуру флюида после смешения с газом
    ' с учетом снижения температуры газа за счет эффекта Джоуля Томсона и калориметрического смешивания
    Set fluid_cas_ = fluid.clone
    fluid_cas_.q_liq_sm3day = 0
    fluid_cas_.q_gas_free_sm3day = q_gas_glv_sm3day
    
    Call fluid.calc_PVT(p_intake_atma, t_intake_C)
    Call fluid_cas_.calc_PVT(p_gas_intake_atma, t_intake_C)
    
    dt_gas_glv_C = fluid_cas_.cJT_Katm * (p_gas_intake_atma - p_intake_atma)
    t_mix_intake_C = feed_t_mix_C(fluid.wm_kgsec, fluid_cas_.wm_kgsec, fluid.cmix_JkgC, fluid_cas_.cg_JkgC, t_intake_C, t_intake_C + dt_gas_glv_C)
    
    ' 4. расчет распределения давления в НКТ
    Set pipe_tubing_above_al.fluid = fluid_al_
    If fast Then
        res1 = pipe_tubing_above_al.calc_dPipe(p_intake_atma, t_mix_intake_C, noCurves)
    Else
        res1 = pipe_tubing_above_al.calc_dPipe(p_intake_atma, t_mix_intake_C, mainCurves)
    End If
    
    p_buf_atma = res1.p_atma
    t_buf_C = res1.t_C
    
    calc_gl_well_from_pwf = True
End Function
Private Sub build_crv(Optional esp_calc As Boolean = False)
    Set crv_p = New CInterpolation
    Set crv_t = New CInterpolation

' pipe generation function from the standard data set
' includes and parameters of the flow in the pipe
' needed to simplify user functions
Private Function new_pipeline4( _
        Optional ByVal trajectory As String = "", _
        Optional ByVal diam_json As String = "", _
        Optional ByVal t_model As String = "", _
        Optional ByVal flow_correlation As Integer = 0, _
        Optional ByVal calibr_grav As Double = 1, _
        Optional ByVal calibr_fric As Double = 1, _
        Optional ByVal h_start_m As Double = -10000000000.1, _
        Optional ByVal h_end_m As Double = 10000000000.1, _
        Optional ByVal h_pump_mes_m As Double = -1, _
        Optional ByVal h_dyn_mes_m As Double = -1, _
        Optional ByVal calc_along_coord As Boolean = True, _
        Optional ByVal flow_along_coord As Boolean = True _
        ) As CPipe
    Dim pipe As New CPipe
On Error GoTo err1:
    Call pipe.init_from_dict(make_construction_dict(trajectory, diam_json), _
                             make_t_model_dict(t_model), _
                             h_start_m, h_end_m, h_pump_mes_m, h_dyn_mes_m)
    Call pipe.set_calc_flow_param(calc_along_coord:=calc_along_coord, _
                                  flow_along_coord:=flow_along_coord, _
                                  hcor:=flow_correlation, _
                                  temp_method:=pipe.param.temp_method)
    pipe.calibr_grav = calibr_grav
    pipe.calibr_fric = calibr_fric
    Set new_pipeline4 = pipe
    Exit Function
err1:
    add_log_msg "Error:new_pipeline3:" & Err.Description
    Err.Raise Err.Number, Err.source, Err.Description
End Function
Private Function make_construction_dict(Optional ByVal trajectory As String = "", _
                                        Optional ByVal diam_json As String = "")
                                        
    Dim dict_constr As New Dictionary
    Dim key As Variant
    Dim diamd As Dictionary
    
    If trajectory = "" Then
        trajectory = encode_pipe_trajectory()
    Else
        If VBA.Mid(trajectory, 1, 1) <> "{" Then
            trajectory = encode_pipe_trajectory(trajectory)
        End If
    End If
    
    dict_constr.Add const_name_h_list_m, ParseJson(trajectory)
    
    If diam_json = "" Then
        diam_json = encode_pipe_diam()
    Else
        If VBA.Mid(diam_json, 1, 1) <> "{" Then
            diam_json = encode_pipe_diam(diam_json)
        End If
    End If
    Set diamd = ParseJson(diam_json)
    For Each key In diamd.keys
        dict_constr.Add key, ParseJson(diam_json)(key)
    Next key
    Set make_construction_dict = dict_constr
End Function
Private Function make_t_model_dict(Optional ByVal t_model As String = "")
    If t_model = "" Then
        t_model = encode_t_model()
    Else
        If VBA.Mid(t_model, 1, 1) <> "{" Then
            t_model = encode_t_model(0, t_model, t_model, t_model)
        End If
    End If
    
    Set make_t_model_dict = ParseJson(t_model)
End Function
Private Function GLV_q_gas_two_valves_sm3day(d_glv_mm As Double, d_surf_mm As Double, _
                                            p_in_atma As Double, _
                                            p_out_atma As Double, _
                                            gamma_g As Double, _
                                            t_surf_C As Double, t_glv_C As Double, _
                                            pipe As CPipe, _
                                            calibr_surf As Double, calibr_glv As Double)
' d_port_mm - диаметр основного порта клапана, мм
' d_vkr_mm  - эффективный диаметр вкруток на выходе, мм
' p_in_atma   - давление на входе в клапан (затруб), атма
' p_out_atma   - давление на выходе клапана (НКТ), атма
' gamma_g   - удельная плотность газа
' t_C       - температура клапана, С
'description_end
    Dim prm As New CSolveParam
    Dim CoeffA(0 To 9)
    Dim pv_atma As Double
    Dim q_gas_sm3day As Double
    Dim res1
    CoeffA(0) = d_glv_mm
    CoeffA(1) = d_surf_mm
    CoeffA(2) = p_in_atma
    CoeffA(3) = p_out_atma
    CoeffA(4) = gamma_g
    CoeffA(5) = t_surf_C
    CoeffA(6) = t_glv_C
    Set CoeffA(7) = pipe
    CoeffA(8) = calibr_surf
    CoeffA(9) = calibr_glv
    
    prm.y_tolerance = 10
    Call solve_equation_bisection("calc_dq_gas_two_valves", p_out_atma, p_in_atma, CoeffA, prm)
    pv_atma = prm.x_solution
'    Debug.Print prm.iterations
    res1 = GLV_q_gas_sm3day(d_surf_mm, p_in_atma, pv_atma, gamma_g, t_surf_C)
    q_gas_sm3day = res1(0)(0)
    GLV_q_gas_two_valves_sm3day = q_gas_sm3day
End Function
' вспомогательная функция для расчета скважины
' вычисляет невязку по давлению на устье от величины
' забойного давления
Public Function calc_well_gl_dpwh_from_pwf(ByVal p_wf_atma As Double, _
                                         CoeffA As Variant) As Double
    Dim p_wh_atma As Double
    Dim t_wf_C As Double
    Dim q_gas_glv_sm3day As Double
    Dim p_cas_atma As Double
    
    p_wh_atma = CoeffA(0)
    t_wf_C = CoeffA(1)
    q_gas_glv_sm3day = CoeffA(2)
    p_cas_atma = CoeffA(3)
    
    Call calc_gl_well_from_pwf(p_wf_atma, _
                               t_wf_C, _
                               p_cas_atma, _
                               q_gas_glv_sm3day)
    
    calc_well_gl_dpwh_from_pwf = Me.p_buf_atma - p_wh_atma
End Function
Public Function calc_gl_well_from_pwh(ByVal p_wh_atma As Double, _
                                      ByVal t_wf_C As Double, _
                                      ByVal p_cas_atma As Double, _
                                      ByVal q_gas_glv_sm3day As Double)
    Dim p_wf_max As Double
    Dim CoeffA(0 To 3)
    Dim prm As New CSolveParam
    
    
    p_wf_max = 500
    
' 0. если задана IPR  то пытаемся оценить граничное забойное давление
    If ipr_vogel.p_res_atma > 0 Then
        p_wf_max = ipr_vogel.p_res_atma
    End If
    
    
    CoeffA(0) = p_wh_atma
    CoeffA(1) = t_wf_C
    CoeffA(2) = q_gas_glv_sm3day
    CoeffA(3) = p_cas_atma
    
    prm.y_tolerance = 0.01
    
    If solve_equation_bisection("calc_well_gl_dpwh_from_pwf", 1, p_wf_max, CoeffA, prm, Me) Then
        Call calc_gl_well_from_pwf(prm.x_solution, _
                                   t_wf_C, _
                                   p_cas_atma, _
                                   q_gas_glv_sm3day)
    End If
End Function
