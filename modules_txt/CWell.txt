Option Explicit

Public d_gas_surf_mm As Double ' диаметр управляющей линии подачи газа на поверхности
Public calibr_glv As Double
Public calibr_surf As Double
Public calibr_grav As Double
Public calibr_fric As Double
Public p_wf_atma As Double
Public t_wf_C As Double

Public t_gas_intake_C As Double
Public dt_gas_glv_C As Double      ' перепад температуры газа на газлифтном клапана за счет эффекта Джоуля Томсона

Private Sub Class_Initialize()
    fast = False
    d_glv_mm = 0   ' 0 означает расчет выключен
    calibr_glv = 1
    d_gas_surf_mm = 0
    calibr_surf = 1
    calibr_grav = 1
    calibr_fric = 1
End Sub
Public Property Get fluid_res() As CPVT
    Set fluid_res = fluid_res_
End Property
Public Property Get fluid_al() As CPVT
    Set fluid_al = fluid_al_
End Property
Public Property Get fluid_cas() As CPVT
    Set fluid_cas = fluid_cas_
End Property
' инициализация конструкции газлифтной скважины
Public Sub init_gl_well(Optional ByVal trajectory As String = "1500", _
                        Optional ByVal d_tub_mm As String = "62", _
                        Optional ByVal d_cas_mm As String = "125", _
                        Optional ByVal t_model As String = "80", _
                        Optional ByVal h_perf_m As Double = 1500, _
                        Optional ByVal h_glv_m As Double = 1200, _
                        Optional ByVal glv_json As String = "", _
                        Optional ByVal IPR_json As String = "", _
                        Optional ByVal calibr_grav As Double = 1, _
                        Optional ByVal calibr_fric As Double = 1, _
                        Optional ByVal flow_corr As Integer = 0)
    If Len(IPR_json) > 0 Then
        Call ipr_vogel.init_json(IPR_json)
        Set ipr_vogel.fluid = fluid
    End If
    
    ' создаем трубу ниже мех. добычи для расчета снизу вверх
    Set pipe_below_al = new_pipeline4(trajectory, _
                                      d_cas_mm, _
                                      t_model, _
                                      flow_corr, _
                                      calibr_grav, _
                                      calibr_fric, _
                                      h_glv_m, h_perf_m, h_glv_m, _
                                      calc_along_coord:=False, _
                                      flow_along_coord:=False)
    Set pipe_below_al.fluid = fluid
    ' создаем трубу НКТ выше мех. добычи для расчета снизу вверх
    Set pipe_tubing_above_al = new_pipeline4(trajectory, _
                                             d_tub_mm, _
                                             t_model, _
                                             flow_corr, _
                                             calibr_grav, _
                                             calibr_fric, _
                                             0, _
                                             h_glv_m, h_glv_m, h_glv_m, _
                                             calc_along_coord:=False, _
                                             flow_along_coord:=False)
    Set pipe_tubing_above_al.fluid = fluid
    'построим распределение давления для газа
    'планируем расчет газа в затрубе сверху вниз
    Set pipe_casing_gas = new_pipeline4(trajectory, _
                                        pipe_diam_subtract(d_cas_mm, d_tub_mm, 5), _
                                        t_model, _
                                        10, _
                                        calibr_grav, _
                                        calibr_fric, _
                                        0, h_glv_m, h_glv_m, h_glv_m, _
                                        calc_along_coord:=True, _
                                        flow_along_coord:=False)
    Set pipe_casing_gas.fluid = fluid
    Me.calibr_grav = calibr_grav
    Me.calibr_fric = calibr_fric
    
    t_surf_C = pipe_casing_gas.ambient_formation.amb_temp_C(0)
                                             
End Sub
' расчет газлифтной скважины снизу вверх по забойному давлению
Public Function calc_gl_well_from_pwf(ByVal p_wf_atma As Double, _
                                      ByVal t_wf_C As Double, _
                             Optional ByVal p_cas_atma As Double = 150, _
                             Optional ByVal q_gas_glv_sm3day As Double = 0)
    Dim res1 As PTtype
    
    calc_gl_well_from_pwf = False
    
    Me.p_wf_atma = p_wf_atma
    Me.t_wf_C = t_wf_C
    
    ' оценим дебит жидкости
    If ipr_vogel.p_res_atma > 0 Then
        q_liq_sm3day = ipr_vogel.calc_q_liq_sm3day(p_wf_atma)
    Else
        q_liq_sm3day = fluid.q_liq_sm3day
    End If
    
    ' рассчитаем участок ниже мех добычи (газлифтного клапана)
    Set pipe_below_al.fluid = fluid
    If fast Then
        res1 = pipe_below_al.calc_dPipe(p_wf_atma, t_wf_C, noCurves)
    Else
        res1 = pipe_below_al.calc_dPipe(p_wf_atma, t_wf_C, mainCurves)
    End If
    p_intake_atma = res1.p_atma
    t_intake_C = res1.t_C
    
    ' рассчитаем распределение давления газа в затрубе
    
    If d_glv_mm > 0 Then
        ' если есть диаметр клапана пытаемся его учесть в расчете
        If d_gas_surf_mm > 0 Then
            q_gas_glv_sm3day = GLV_q_gas_two_valves_sm3day(d_glv_mm, d_gas_surf_mm, _
                                                           p_cas_atma, p_intake_atma, _
                                                           fluid.gamma_g, _
                                                           t_surf_C, t_intake_C, _
                                                           pipe_casing_gas, _
                                                           calibr_surf, calibr_glv)
        Else
            res1 = pipe_casing_gas.calc_dPipe(p_cas_atma, t_intake_C, mainCurves)
            q_gas_glv_sm3day = GLV_q_gas_sm3day(d_glv_mm, res1.p_atma, p_intake_atma, fluid.gamma_g, t_intake_C)(0)(0)
        End If
    Else
        res1 = pipe_casing_gas.calc_dPipe(p_cas_atma, t_intake_C, mainCurves)
    End If
    p_gas_intake_atma = res1.p_atma
    t_gas_intake_C = res1.t_C
    
    ' в итого подготовим флюид для расчета выше клапана
    Set fluid_al_ = fluid.clone
    fluid_al_.q_gas_free_sm3day = fluid_al_.q_gas_free_sm3day + q_gas_glv_sm3day
    Me.q_gas_glv_sm3day = q_gas_glv_sm3day
    ' оценим температуру флюида после смешения с газом
    ' с учетом снижения температуры газа за счет эффекта Джоуля Томсона и калориметрического смешивания
    Set fluid_cas_ = fluid.clone
    fluid_cas_.q_liq_sm3day = 0
    fluid_cas_.q_gas_free_sm3day = q_gas_glv_sm3day
    
    Call fluid.calc_PVT(p_intake_atma, t_intake_C)
    Call fluid_cas_.calc_PVT(p_gas_intake_atma, t_intake_C)
    
    dt_gas_glv_C = fluid_cas_.cJT_Katm * (p_gas_intake_atma - p_intake_atma)
    t_mix_intake_C = feed_t_mix_C(fluid.wm_kgsec, fluid_cas_.wm_kgsec, fluid.cmix_JkgC, fluid_cas_.cg_JkgC, t_intake_C, t_intake_C + dt_gas_glv_C)
    
    ' 4. расчет распределения давления в НКТ
    Set pipe_tubing_above_al.fluid = fluid_al_
    If fast Then
        res1 = pipe_tubing_above_al.calc_dPipe(p_intake_atma, t_mix_intake_C, noCurves)
    Else
        res1 = pipe_tubing_above_al.calc_dPipe(p_intake_atma, t_mix_intake_C, mainCurves)
    End If
    
    p_buf_atma = res1.p_atma
    t_buf_C = res1.t_C
    
    calc_gl_well_from_pwf = True
End Function
Private Sub build_crv()
    Set crv_p = New CInterpolation
    Set crv_t = New CInterpolation
    Set crv_t_amb = New CInterpolation
    
    Call crv_p.AddPointsCurve(pipe_tubing_above_al.curve("c_P").ClonePointsToNum(25), pipe_below_al.curve("c_P").ClonePointsToNum(10))
    crv_p.xName = "h_mes,m"
    crv_p.yName = "p,atma"
    Call crv_t.AddPointsCurve(pipe_tubing_above_al.curve("c_T").ClonePointsToNum(25), pipe_below_al.curve("c_T").ClonePointsToNum(10))
    crv_t.xName = "h_mes,m"
    crv_t.yName = "t,C"
    Call crv_t_amb.AddPointsCurve(pipe_tubing_above_al.curve("c_Tamb").ClonePointsToNum(25), pipe_below_al.curve("c_Tamb").ClonePointsToNum(10))
    crv_t.xName = "h_mes,m"
    crv_t.yName = "t_amb,C"
    
    Set crv_cas_p = pipe_casing_gas.curve("c_P").ClonePointsToNum(25)
End Sub
Public Function get_result_dict()
        
    Dim dout As New Dictionary
    Call build_crv
    
    dout.Add "q_liq_sm3day", Round(q_liq_sm3day, 2)
    dout.Add "p_wf_atma", Round(p_wf_atma, 2)
    dout.Add "p_intake_atma", Round(p_intake_atma, 2)
    dout.Add "p_gas_intake_atma", Round(p_gas_intake_atma, 2)
    dout.Add "p_buf_atma", Round(p_buf_atma, 2)
    dout.Add "crv_p", crv_p.getDict
    dout.Add "crv_t", crv_t.getDict
    dout.Add "crv_p_cas", crv_cas_p.getDict
    dout.Add "crv_t_amb", crv_t_amb.getDict
    dout.Add "q_gas_cas_sm3day", Round(q_gas_glv_sm3day, 2)
    dout.Add "t_intake_C", Round(t_intake_C, 2)
    dout.Add "tmix", Round(t_mix_intake_C, 2)
    dout.Add "dtgas", Round(dt_gas_glv_C, 2)
    dout.Add "t_gas_intake", Round(t_intake_C + dt_gas_glv_C, 2)
    Set get_result_dict = dout
End Function
Public Function get_result_json()
    get_result_json = ConvertToJson(get_result_dict)
End Function
' pipe generation function from the standard data set
' includes and parameters of the flow in the pipe
' needed to simplify user functions
Private Function new_pipeline4( _
        Optional ByVal trajectory As String = "", _
        Optional ByVal diam_json As String = "", _
        Optional ByVal t_model As String = "", _
        Optional ByVal flow_correlation As Integer = 0, _
        Optional ByVal calibr_grav As Double = 1, _
        Optional ByVal calibr_fric As Double = 1, _
        Optional ByVal h_start_m As Double = -10000000000.1, _
        Optional ByVal h_end_m As Double = 10000000000.1, _
        Optional ByVal h_pump_mes_m As Double = -1, _
        Optional ByVal h_dyn_mes_m As Double = -1, _
        Optional ByVal calc_along_coord As Boolean = True, _
        Optional ByVal flow_along_coord As Boolean = True _
        ) As CPipe
    Dim pipe As New CPipe
On Error GoTo err1:
    Call pipe.init_from_dict(make_construction_dict(trajectory, diam_json), _
                             make_t_model_dict(t_model), _
                             h_start_m, h_end_m, h_pump_mes_m, h_dyn_mes_m)
    Call pipe.set_calc_flow_param(calc_along_coord:=calc_along_coord, _
                                  flow_along_coord:=flow_along_coord, _
                                  hcor:=flow_correlation, _
                                  temp_method:=pipe.param.temp_method)
    pipe.calibr_grav = calibr_grav
    pipe.calibr_fric = calibr_fric
    Set new_pipeline4 = pipe
    Exit Function
err1:
    add_log_msg "Error:new_pipeline3:" & Err.Description
    Err.Raise Err.Number, Err.source, Err.Description
End Function
Private Function make_construction_dict(Optional ByVal trajectory As String = "", _
                                        Optional ByVal diam_json As String = "")
                                        
    Dim dict_constr As New Dictionary
    Dim key As Variant
    Dim diamd As Dictionary
    
    If trajectory = "" Then
        trajectory = encode_pipe_trajectory()
    Else
        If VBA.Mid(trajectory, 1, 1) <> "{" Then
            trajectory = encode_pipe_trajectory(trajectory)
        End If
    End If
    
    dict_constr.Add const_name_h_list_m, ParseJson(trajectory)
    
    If diam_json = "" Then
        diam_json = encode_pipe_diam()
    Else
        If VBA.Mid(diam_json, 1, 1) <> "{" Then
            diam_json = encode_pipe_diam(diam_json)
        End If
    End If
    Set diamd = ParseJson(diam_json)
    For Each key In diamd.keys
        dict_constr.Add key, ParseJson(diam_json)(key)
    Next key
    Set make_construction_dict = dict_constr
End Function
Private Function make_t_model_dict(Optional ByVal t_model As String = "")
    If t_model = "" Then
        t_model = encode_t_model()
    Else
        If VBA.Mid(t_model, 1, 1) <> "{" Then
            t_model = encode_t_model(0, t_model, t_model, t_model)
        End If
    End If
    
    Set make_t_model_dict = ParseJson(t_model)
End Function
Private Function GLV_q_gas_two_valves_sm3day(d_glv_mm As Double, d_surf_mm As Double, _
                                            p_in_atma As Double, _
                                            p_out_atma As Double, _
                                            gamma_g As Double, _
                                            t_surf_C As Double, t_glv_C As Double, _
                                            pipe As CPipe, _
                                            calibr_surf As Double, calibr_glv As Double)
' d_port_mm - диаметр основного порта клапана, мм
' d_vkr_mm  - эффективный диаметр вкруток на выходе, мм
' p_in_atma   - давление на входе в клапан (затруб), атма
' p_out_atma   - давление на выходе клапана (НКТ), атма
' gamma_g   - удельная плотность газа
' t_C       - температура клапана, С
'description_end
    Dim prm As New CSolveParam
    Dim CoeffA(0 To 9)
    Dim pv_atma As Double
    Dim q_gas_sm3day As Double
    Dim res1
    CoeffA(0) = d_glv_mm
    CoeffA(1) = d_surf_mm
    CoeffA(2) = p_in_atma
    CoeffA(3) = p_out_atma
    CoeffA(4) = gamma_g
    CoeffA(5) = t_surf_C
    CoeffA(6) = t_glv_C
    Set CoeffA(7) = pipe
    CoeffA(8) = calibr_surf
    CoeffA(9) = calibr_glv
    
    prm.y_tolerance = 10
    Call solve_equation_bisection("calc_dq_gas_two_valves", p_out_atma, p_in_atma, CoeffA, prm)
    pv_atma = prm.x_solution
'    Debug.Print prm.iterations
    res1 = GLV_q_gas_sm3day(d_surf_mm, p_in_atma, pv_atma, gamma_g, t_surf_C)
    q_gas_sm3day = res1(0)(0)
    GLV_q_gas_two_valves_sm3day = q_gas_sm3day
End Function
' вспомогательная функция для расчета скважины
' вычисляет невязку по давлению на устье от величины
' забойного давления
Public Function calc_well_gl_dpwh_from_pwf(ByVal p_wf_atma As Double, _
                                         CoeffA As Variant) As Double
    Dim p_wh_atma As Double
    Dim t_wf_C As Double
    Dim q_gas_glv_sm3day As Double
    Dim p_cas_atma As Double
    
    p_wh_atma = CoeffA(0)
    t_wf_C = CoeffA(1)
    q_gas_glv_sm3day = CoeffA(2)
    p_cas_atma = CoeffA(3)
    
    Call calc_gl_well_from_pwf(p_wf_atma, _
                               t_wf_C, _
                               p_cas_atma, _
                               q_gas_glv_sm3day)
    
    calc_well_gl_dpwh_from_pwf = Me.p_buf_atma - p_wh_atma
End Function
Public Function calc_gl_well_from_pwh(ByVal p_wh_atma As Double, _
                                      ByVal t_wf_C As Double, _
                                      ByVal p_cas_atma As Double, _
                                      ByVal q_gas_glv_sm3day As Double)
    Dim p_wf_max As Double
    Dim CoeffA(0 To 3)
    Dim prm As New CSolveParam
    
    
    p_wf_max = 500
    
' 0. если задана IPR  то пытаемся оценить граничное забойное давление
    If ipr_vogel.p_res_atma > 0 Then
        p_wf_max = ipr_vogel.p_res_atma
    End If
    
    
    CoeffA(0) = p_wh_atma
    CoeffA(1) = t_wf_C
    CoeffA(2) = q_gas_glv_sm3day
    CoeffA(3) = p_cas_atma
    
    prm.y_tolerance = 0.01
    
    If solve_equation_bisection("calc_well_gl_dpwh_from_pwf", 1, p_wf_max, CoeffA, prm, Me) Then
        Call calc_gl_well_from_pwf(prm.x_solution, _
                                   t_wf_C, _
                                   p_cas_atma, _
                                   q_gas_glv_sm3day)
    End If
End Function
